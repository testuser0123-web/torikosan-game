CREATE TABLE posts (
  id SERIAL PRIMARY KEY,

  created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
  img VARCHAR(63),
  user_id VARCHAR(20),
  ip_address VARCHAR(255),
